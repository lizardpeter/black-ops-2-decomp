cmake_minimum_required(VERSION 3.20)
project(t6_xcurl_consumer LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The Store XCurl ABI requires Windows x64")
endif()
set(T6_XCURL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../reconstructed_dlls/xcurl" "${CMAKE_CURRENT_BINARY_DIR}/xcurl")
add_executable(native_xcurl_consumer main.cpp import_probe.cpp)
target_link_libraries(native_xcurl_consumer PRIVATE t6::xcurl_import_abi)
if(MSVC)
    target_link_options(native_xcurl_consumer PRIVATE "/MAP:$<TARGET_FILE_DIR:native_xcurl_consumer>/native_xcurl_consumer.map")
endif()
